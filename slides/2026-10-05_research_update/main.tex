\section{The question}

\begin{frame}{AI can now generate video that never ends}
\begin{columns}[T]
\column{0.40\textwidth}
\begin{itemize}
  \item New \emph{streaming} video models write a video one short chunk at a time, each chunk looking at the last few.
  \item In principle they can run forever: games, live worlds, endless scenes.
  \item We ran three of them for \textbf{3 minutes} on 8 very different prompts.
\end{itemize}
\vspace{4pt}
\take{What happens to a video\\ if you just keep going?}
\column{0.58\textwidth}
\centering
\fig{photo_end_states_top}\\[-1pt]
{\scriptsize The first frame of each of the 8 prompts (Self Forcing).}
\end{columns}
\end{frame}

\begin{frame}{\ldots but they all die the same death}
\centering
\figh{0.66}{photo_sf_converge}\\[2pt]
{\scriptsize One model (Self Forcing), six prompts, frames from 0 to 178 s. Mammoths, waves, a monster, a pigeon, pirate ships: after three minutes they all become the same purple-blue streaks.}
\end{frame}

\begin{frame}{The question, in one sentence}
\begin{columns}[c]
\column{0.42\textwidth}
\take{\large Long videos \emph{decay}. \\[3pt] \normalsize \textbf{Where} do they go, \textbf{how fast}, and \\ \textbf{which design choice} decides it?}
\vspace{2pt}
\begin{itemize}\small
  \item Think of a marble on a landscape: wherever you drop it, it rolls into a valley.
  \item Physicists call such a valley an \textbf{attractor}.
  \item Our claim: video generators have attractors, and each model has its own kind.
\end{itemize}
\column{0.56\textwidth}
\centering
\fig{line_phase_portrait}\\[-1pt]
{\scriptsize Each line is one 3-minute video, drawn in a 2-D map of its content (dot = start, $\times$ = end). Self Forcing: all lines run into one corner. The others stay where they started.}
\end{columns}
\end{frame}

\begin{frame}{Why this matters}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{Everyone hits this wall}
Streaming video, game engines and world models all generate one step at a time. They all drift, wash out or freeze. Each new method patches a symptom.
\end{block}
\begin{block}{Nobody has measured it}
Papers list failure symptoms. No one asks, with numbers, what the end state is, how long the prompt is remembered, or which knob changes the destination.
\end{block}
\column{0.48\textwidth}
\begin{block}{What we provide}
\begin{itemize}\small
  \item A map of each model's \textbf{end states}
  \item A \textbf{memory half-life} for the prompt
  \item \textbf{Sensitivity}: does a tiny nudge change the story?
  \item \textbf{Tipping knobs}: one design change flips the end state
  \item Tools anyone can run on a new model
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\section{How we got here}

\begin{frame}{Where we started: fixing videos at test time}
\begin{columns}[c]
\column{0.38\textwidth}
\begin{itemize}\small
  \item Original project: adapt the model to each video while generating (\emph{test-time adaptation}, TTA).
  \item At 1{,}000-video scale it does \textbf{nothing measurable}: FVD 154.7 $\to$ 153.4, within noise.
  \item An earlier ``+5\% FVD'' claim did not survive scale. It has been removed from the draft.
\end{itemize}
\column{0.60\textwidth}
\centering
\fig{bar_tta_null}\\[-1pt]
{\scriptsize LongCat-Video 13.6B, standard-horizon continuation. Lower FVD is better.}
\end{columns}
\end{frame}

\begin{frame}{33 fixes later: one pattern}
\centering
\figh{0.40}{dot_atlas}\\[2pt]
{\scriptsize Every test-time intervention we tried on Wan2.1 Self/Rolling Forcing, measured against doing nothing. Blue: methods that \emph{edit} the video as it is made. Tan: methods that \emph{pick} among the model's own futures.}
\take{Editing the trajectory can wreck a video (down to $-53$ quality points). Picking among the model's own futures is safe, but only \emph{slows} the decay. So we stopped trying to fix the decay and started measuring it.}
\end{frame}

\section{Approach}

\begin{frame}{Setup}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Models (all on Wan2.1, 1.3B)}
\begin{itemize}\small
  \item \textbf{Self Forcing}: rolling window, no anchor
  \item \textbf{Rolling Forcing}: keeps the first chunk as an anchor (``sink'')
  \item \textbf{LongLive}: sink plus long-video tuning
\end{itemize}
\end{block}
\begin{block}{Rollouts}
\small 8 MovieGen prompts $\times$ 2 seeds $\times$ 3 models = 48 videos of 3 minutes (2{,}877 frames each). Plus 128 real 30\,s clips as a control. About 8 H200-hours so far.
\end{block}
\column{0.50\textwidth}
\begin{block}{Measurements, every frame}
\begin{itemize}\small
  \item \textbf{Content}: DINOv2 embedding (what is in the frame)
  \item \textbf{Look}: brightness, contrast, saturation, colourfulness, sharpness, spectral slope
  \item \textbf{Motion}: frame difference, optical flow
\end{itemize}
\end{block}
\begin{block}{Treat the generator as a dynamical system}
\small The state is the recent frames. The rule is the model. The noise is the random seed.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Toolkit: dynamical-systems questions, video measurements}
\small
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\linewidth}{>{\bfseries}l X X}
\toprule
Concept & Question & Measurement \\
\midrule
Attractor & Where do videos end up? & Distance between videos of different prompts; can the prompt still be identified from the frame? \\
Relaxation & How fast? & Exponential fit to that distance; prompt \emph{half-life} \\
Lyapunov (twins) & Does a tiny nudge change the story? & Re-run with one noise block perturbed; track how fast the twins separate \\
Bifurcation & Which knob flips the end state? & Same weights, change sink size or attention window \\
Recurrence & Is the motion structured or random? & Recurrence analysis vs.\ phase-randomized surrogates \\
Early warning & Can we see collapse coming? & Rising variance and autocorrelation before collapse \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}{Self Forcing has one global attractor}
\centering
\figh{0.42}{line_convergence}\\[2pt]
{\scriptsize (a) Distance between videos of different prompts. Solid: different prompts. Dashed: same prompt, other seed. (b) Share of videos whose prompt can still be identified from the current frame (8 prompts, chance 12.5\%).}
\take{Self Forcing: distance falls 0.97 $\to$ 0.64 and identification falls 100\% $\to$ 34\%. Rolling Forcing and LongLive keep every prompt (100\% and 98\%). This is not shared noise: the seeds differ.}
\end{frame}

\begin{frame}{The prompt has a half-life of about one minute}
\begin{columns}[c]
\column{0.60\textwidth}
\fig{line_halflife}
\column{0.38\textwidth}
\begin{itemize}\small
  \item Distance between different prompts decays exponentially: half-life \textbf{about 1 minute} (58--66\,s depending on the fit window).
  \item It levels off at the \textbf{noise floor}: two seeds of the \emph{same} prompt are as far apart as two different prompts. The prompt is fully erased.
  \item Pull toward the end state is exponential too: $\tau \approx 360$\,s, $R^2 = 0.91$.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{What the decay looks like}
\centering
\figh{0.38}{line_observables}\\[2pt]
{\scriptsize Median over 16 videos per model, band = interquartile range.}
\take{Self Forcing gets darker, more saturated and streaky. Rolling Forcing brightens toward glare. LongLive holds its look. Motion collapses fast (seconds); appearance drifts slowly (minutes).}
\end{frame}

\begin{frame}{Twins: same video, one tiny nudge}
\centering
\figh{0.22}{photo_twins_sf}\\[2pt]
\figh{0.36}{line_twins}\\[-2pt]
{\scriptsize Runs are bit-for-bit reproducible, so the only difference between the twins is one perturbed noise block at 29\,s. (a) Their distance grows at a model-specific rate, whatever the size of the nudge. (b) LongLive is the \emph{most} sensitive, yet its twins end as close in content as Rolling Forcing's. Only Self Forcing's twins end up in a different story.}
\end{frame}

\begin{frame}{One knob flips the attractor}
\centering
\figh{0.34}{photo_bifurcation_p0}\\[1pt]
\figh{0.30}{line_bifurcation}\\[-2pt]
{\scriptsize Same Self Forcing weights; only the inference settings change. Adding a 3-frame sink switches the \emph{global} attractor to \emph{prompt-anchored} fixed points. A shorter window alone changes how the collapse looks, not whether it happens.}
\end{frame}

\begin{frame}{Structure, but no early warning}
\begin{columns}[c]
\column{0.62\textwidth}
\fig{line_ews}
\column{0.36\textwidth}
\fig{bar_rqa}
\end{columns}
\vspace{2pt}
{\scriptsize Left: 16 Self Forcing videos aligned at their collapse time. The transition from 10\% to 90\% takes about 90\,s. Classic warning signals (rising autocorrelation) are no more common before collapse than in videos that never collapse. Right: every model's motion is more structured than shuffled surrogates (100\% of runs).}
\take{Collapse is a slow ramp, not a tipping point. That is a negative result we will report.}
\end{frame}

\begin{frame}{Checking a rival claim: periodic ``snap-back''}
\centering
\figh{0.36}{line_lol}\\[2pt]
{\scriptsize LoL (arXiv 2601.16914) reports LongLive snapping back to its first frames at the same moments in every video. We test for synchronized returns toward the opening across the 16 runs per model, against 500 time-shifted surrogates (red line = observed).}
\take{No synchronized snap-back within 3 minutes for any model ($p \geq 0.8$). LoL studies hour-long videos, so a longer LongLive run is still open.}
\end{frame}

\begin{frame}{Has anyone done this already?}
\small
About 60 targeted searches (2024--Oct 2026): \textbf{no paper treats streaming video generators as dynamical systems}. The symptoms are well documented. The characterization is new.
\vspace{4pt}
\renewcommand{\arraystretch}{1.15}
\begin{tabularx}{\linewidth}{l X l}
\toprule
Closest work & What it shows & Overlap \\
\midrule
LoL (2601.16914) & Sink-collapse loops at hour scale & partial: test longer \\
Deep / Rolling Sink, FreqForcing & A sink slows drift & partial: we show a \emph{type switch} \\
Effective-rank collapse (2607.27036) & Abrupt representation collapse at drift onset & reconcile with ``slow ramp'' \\
Meta-ARVDM (2503.10704) & Theory of history forgetting & conceptual, frames not prompt \\
Hintze et al., Patterns 2025 & Image loops collapse to $\sim$12 attractors & concept, not video \\
\bottomrule
\end{tabularx}
\vspace{3pt}
\take{Safe claims: attractor types, prompt half-life, twins on video, RQA plus no early warning, real-video control. Avoid: ``sinks prevent drift'' (known), ``first dynamical view of generative models'' (too broad).}
\end{frame}

\section{Next}

\begin{frame}{Next two weeks}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{Scale and sharpen (about 10 H200-hours)}
\begin{itemize}\small
  \item 8 $\to$ 32 prompts for every result above
  \item Sink sizes 1 and 2: where exactly does the switch happen?
  \item Rolling Forcing \emph{without} its sink: does it collapse like Self Forcing?
  \item One long LongLive run (beyond 3 min) for the snap-back test
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Paper (CVPR 2027, deadline Nov 16)}
\begin{itemize}\small
  \item Lead with the attractor map and the one-knob switch
  \item Supporting chapters: the TTA null at scale and the 33-intervention atlas
  \item Release the toolkit as a drop-in diagnostic
\end{itemize}
\end{block}
\end{columns}
\vspace{6pt}
\take{\textbf{Takeaway.} Long videos do not just ``get worse''. Each model falls into its own kind of attractor, at a measurable rate, and a single design knob can change which kind.}
\end{frame}
