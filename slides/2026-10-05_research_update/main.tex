\section{The question}

\begin{frame}{AI can now generate video that never ends}
\begin{columns}[T]
\column{0.40\textwidth}
\begin{itemize}
  \item New \emph{streaming} video models write a video one short chunk at a time, each chunk looking at the last few.
  \item In principle they can run forever: games, live worlds, endless scenes.
  \item We ran three of them for \textbf{3 minutes} on 8 very different prompts.
\end{itemize}
\vspace{4pt}
\take{What happens to a video\\ if you just keep going?}
\column{0.58\textwidth}
\centering
\fig{photo_end_states_top}\\[-1pt]
{\scriptsize The first frame of each of the 8 prompts (Self Forcing).}
\end{columns}
\end{frame}

\begin{frame}{\ldots but on Self Forcing, they all die the same death}
\centering
\figh{0.66}{photo_sf_converge}\\[2pt]
{\scriptsize One model (Self Forcing), six prompts, frames from 0 to 178 s. Mammoths, waves, a monster, a pigeon, pirate ships: after three minutes they all become the same purple-blue streaks.}
\end{frame}

\begin{frame}{The question, in one sentence}
\take{\large Long videos \emph{decay}. \textbf{Where} do they go, \textbf{how fast}, and \textbf{which design choice} decides it?}
\vspace{-6pt}
\centering
\figh{0.58}{line_phase_portrait}\\[-1pt]
{\scriptsize We use \emph{attractor} operationally: a region of representation space where trajectories from different prompts converge and stay. Self Forcing's videos converge to one region; the others stay apart. The 2-D map keeps 19\% of the variation; slide 13 measures in the full embedding.}
\end{frame}

\begin{frame}{Why this matters}
\vspace{-6pt}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{A common problem, with very different outcomes}
\footnotesize Streaming video, game engines and world models generate one step at a time from their own output. Long rollouts often accumulate drift, and recent designs (sinks, long-video tuning) reduce it a lot. In our runs, how a video ends depends strongly on the model and its inference settings.
\end{block}
\begin{block}{Measured as symptoms, not as dynamics}
\footnotesize Benchmarks and recent papers already quantify degradation: quality and consistency over long videos (VBench-Long), colour and motion drift, spectral drift, representation collapse, sink collapse. What we have not found is a characterization of the \emph{dynamics}: where trajectories converge, how fast the prompt is lost, how perturbations grow, which design change alters the end state.
\end{block}
\column{0.48\textwidth}
\begin{block}{What we provide}
\begin{itemize}\footnotesize
  \item A map of each model's \textbf{end states}
  \item A \textbf{memory half-life} for the prompt
  \item \textbf{Perturbation growth}: does a tiny nudge change the story?
  \item \textbf{Regime-changing settings}: one memory setting changes the long-run regime
  \item Tools anyone can run on a new model
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\section{How we got here}

\begin{frame}{Where we started: fixing videos at test time}
\begin{columns}[c]
\column{0.38\textwidth}
\begin{itemize}\small
  \item Original project: adapt the model to each video while generating (\emph{test-time adaptation}, TTA).
  \item At 1{,}000-video scale it does \textbf{nothing measurable}: FVD 154.7 $\to$ 153.4, within noise.
  \item An earlier ``+5\% FVD'' claim did not survive scale. It has been removed from the draft.
\end{itemize}
\column{0.60\textwidth}
\centering
\fig{bar_tta_null}\\[-1pt]
{\scriptsize LongCat-Video 13.6B, standard-horizon continuation. Lower FVD is better.}
\end{columns}
\end{frame}

\begin{frame}{33 fixes later: one pattern}
\centering
\figh{0.40}{dot_atlas}\\[2pt]
{\scriptsize Test-time interventions on Wan2.1 Self/Rolling Forcing, each compared with doing nothing on the same clips. Blue: 11 families of methods that \emph{edit} the video as it is made (one point per family, the median of its variants). Tan: 11 results from methods that \emph{pick} among the model's own futures.}
\take{Editing the trajectory can wreck a video (down to $-28$ quality points). Picking among the model's own futures is safe, but only \emph{slows} the decay. So we stopped trying to fix the decay and started measuring it.}
\end{frame}

\section{Approach}

\begin{frame}{Setup}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Models (all on Wan2.1, 1.3B)}
\begin{itemize}\small
  \item \textbf{Self Forcing}: rolling window, no anchor
  \item \textbf{Rolling Forcing}: keeps the first chunk as an anchor (``sink'')
  \item \textbf{LongLive}: sink plus long-video tuning
\end{itemize}
\end{block}
\begin{block}{Rollouts}
\small 8 MovieGen prompts $\times$ 2 seeds $\times$ 3 models = 48 videos of 3 minutes (2{,}877 frames each). Plus 128 real 30\,s clips as a control. About 8 H200-hours so far.
\end{block}
\column{0.50\textwidth}
\begin{block}{Measurements, every frame}
\begin{itemize}\small
  \item \textbf{Content}: DINOv2 embedding (what is in the frame)
  \item \textbf{Look}: brightness, contrast, saturation, colourfulness, sharpness, spectral slope
  \item \textbf{Motion}: frame difference, optical flow
\end{itemize}
\end{block}
\begin{block}{A prompt-conditioned stochastic dynamical system}
\small $X_{t+1} = F_{\theta,\lambda}(X_t;\, c,\, \xi_t)$\\[2pt]
\footnotesize $X_t$: state (recent latent frames and their memory cache) $\cdot$ $F_{\theta,\lambda}$: the generator, weights $\theta$, inference settings $\lambda$ (sink, window) $\cdot$ $c$: prompt $\cdot$ $\xi_t$: generation noise. We ask how the distribution of $X_t$ given $c$ evolves.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Toolkit: questions about the dynamics, and how we measure them}
\small
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\linewidth}{>{\bfseries}l X X}
\toprule
Concept & Question & Measurement \\
\midrule
Attractor & Where do trajectories concentrate? & Cross-prompt distances; can the prompt still be recovered from the frame? \\
Relaxation & How fast is conditioning information lost? & Exponential fit to that distance; prompt \emph{half-life} \\
Perturbation growth & How does a small local change propagate? & Twin rollouts, one noise block nudged; finite-time growth rate \\
Regime transition & Which settings change long-run behaviour? & Same weights; change sink size and attention window \\
Recurrence & Does temporal ordering carry structure? & Recurrence analysis vs.\ phase-randomized surrogates \\
Predictability & Can failure be anticipated? & Pre-collapse variance and autocorrelation statistics \\
\bottomrule
\end{tabularx}
\vspace{4pt}
\footnotesize Physics terms describe behaviour measured in representation space, not proven properties of the model's full internal state.
\end{frame}

\begin{frame}{Self Forcing: a shared, prompt-independent attracting regime}
\centering
\figh{0.49}{line_convergence}\\[-4pt]
\begin{columns}[T]
\column{0.49\textwidth}
\tiny\raggedright
\textbf{(a)} Every frame (16/s): resized to 224\,px short side, centre-cropped 224$\times$224, encoded by \textbf{DINOv2 ViT-B/14} (\texttt{dinov2\_vitb14}); final-layer [CLS] vector, 768-d, averaged per second and normalised. Plotted: mean \textbf{cosine distance} $1-\cos(z_i,z_j)$ at each second. \textbf{Solid}: 56 pairs differing in prompt \emph{and} seed. \textbf{Dashed} control: 8 pairs with the same prompt, different seed (the spread randomness alone produces).
\column{0.49\textwidth}
\tiny\raggedright
\textbf{(b)} \textbf{Nearest-neighbour} matching on the same vectors; \textbf{no classifier is trained}. Query: one video at second $t$. Candidates: the 8 videos of the \emph{other} seed at the same second, one per prompt. Correct if the most cosine-similar candidate has the query's prompt. 16 queries per second (8 prompts $\times$ 2 seeds). \textbf{Chance = 1/8 = 12.5\%}.
\end{columns}
\vspace{2pt}
{\tiny\raggedright\textbf{Data}, same for all three models: 8 MovieGen prompts $\times$ seeds 0 and 1 = 16 videos of 178--179\,s. Within a seed all prompts get identical noise, so both panels compare only across seeds. Bands: 95\% bootstrap over prompts. No smoothing beyond the 1-s average.\par}
\vspace{3pt}
{\footnotesize\textbf{Shared attracting regime:} Self Forcing's \emph{different} prompts become nearly as alike as two seeds of the \emph{same} prompt (0.97 $\to$ 0.63 vs.\ 0.57), and a frame stops revealing its prompt (100\% $\to$ 34\%). Rolling Forcing and LongLive keep prompts apart (0.97) and identifiable (100\%, 98\%). The model's own latents converge too (0.92 $\to$ 0.43 across seeds).\par}
\end{frame}

\begin{frame}{The prompt has a half-life of about one minute}
\begin{columns}[c]
\column{0.60\textwidth}
\fig{line_halflife}
\column{0.38\textwidth}
\begin{itemize}\small
  \item The cross-prompt distance in DINOv2 space decays exponentially. \textbf{Half-life} = time for its excess above the fitted long-run level to halve: \textbf{about 1 minute} (58--72\,s, depending on fit window and pairs averaged).
  \item It approaches the \textbf{noise floor}: by 3 minutes two different prompts (0.63) are nearly as far apart as two seeds of the \emph{same} prompt (0.57). In this signal, the prompt is almost fully erased.
  \item Pull toward the end state is exponential too: $\tau \approx 360$\,s, $R^2 = 0.91$.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{What the decay looks like}
\centering
\figh{0.56}{line_observables}\\[2pt]
{\scriptsize Low-level measurements per frame, averaged per second, for the 16 videos of each model (8 prompts $\times$ 2 seeds).}
\take{Self Forcing gets darker, more saturated and streaky. Rolling Forcing brightens toward glare. LongLive holds its look. Motion collapses fast (seconds); appearance drifts slowly (minutes).}
\end{frame}

\begin{frame}{Twins: same video, one tiny nudge}
\centering
\figh{0.38}{photo_twins_sf}\\[4pt]
{\scriptsize Self Forcing, ``Tokyo street''. The runs are bit-for-bit reproducible, so the only difference between the two rows is one block of input noise nudged at 29\,s. Identical at 20 and 30\,s, slightly different by 60\,s, telling different stories by 120\,s.}
\take{Does a tiny nudge change the story? For Self Forcing, yes.}
\end{frame}

\begin{frame}{Twins: local perturbation growth does not predict long-run divergence}
\centering
\figh{0.48}{line_twins}\\[2pt]
{\scriptsize All 8 prompts per model, seed 0. $\varepsilon$ = relative size of the nudge to one noise block. (a) Content distance between twins over time, log scale; the twins are identical before 29\,s. (b) Finite-time perturbation growth rate: how fast the gap grows in the model's latent space in the first 7.5\,s; the same for both nudge sizes. Noise after the nudge is identical in both twins. (c) Content gap in the last 10\,s.}
\take{LongLive has the fastest local perturbation growth, but not the largest long-run divergence; only Self Forcing's twins end up in different states. Local sensitivity and long-run degradation are different properties.}
\end{frame}

\begin{frame}{Changing memory alone changes the long-run regime}
\centering
\figh{0.52}{photo_bifurcation_p0}\\[3pt]
{\scriptsize Same Self Forcing weights, same prompt and seed; only two inference settings change, both counted in latent frames (1 latent frame = 4 video frames = 0.25\,s). \emph{Window}: how far back the model attends (21 $\approx$ 5.3\,s, 12 = 3\,s). \emph{Sink}: the first latent frames kept in memory permanently (3 = the first 0.75\,s).}
\take{Weights fixed. A sink of 3 latent frames keeps the scene for 3 minutes. A shorter window alone still collapses, into a different pattern.}
\end{frame}

\begin{frame}{Changing memory alone changes the long-run regime: all 8 prompts}
\centering
\figh{0.46}{line_bifurcation}\\[2pt]
{\scriptsize Same four settings, 8 prompts at seed 0 (the 28 prompt pairs, which share the same input noise; on the main runs, same-seed and cross-seed pairs give the same curve, 0.636 vs.\ 0.628 at 3 min). (a) Mean cosine distance between DINOv2 [CLS] vectors of different prompts. (b) Cosine similarity of each video to its own first 2\,s.}
\take{Without a sink, prompts merge and openings are forgotten (distance $\to$ 0.67, similarity $\to$ 0.2). With a sink, prompts stay apart ($>$0.9) and openings are kept ($\approx$0.7). Holding the weights fixed, the memory setting decides whether prompts merge: a \emph{regime transition}. Two sink sizes have been tested so far; a dense sweep would locate where the switch occurs.}
\end{frame}

\begin{frame}{Can pre-collapse statistics predict failure? Not with these}
\centering
\figh{0.58}{line_ews}\\[2pt]
{\scriptsize (a) Each line is one Self Forcing video, aligned at its collapse (first second after which it stays closer to the shared end state than to its opening for 10\,s): a slow ramp of about 90\,s from 10\% to 90\%. (b) Share of videos whose lag-1 autocorrelation rises in the 60\,s before collapse (blue) vs.\ the same window in models that never collapse (grey).}
\take{We do not find a useful pre-collapse warning in the tested autocorrelation and variance statistics.}
\end{frame}

\begin{frame}{Temporal ordering adds recurrence structure beyond the frequency spectrum}
\begin{columns}[T]
\column{0.70\textwidth}
\figh{0.58}{bar_rqa_gap}
\column{0.28\textwidth}
\scriptsize\raggedright
\textbf{Recurrence determinism (DET):} share of revisits to an earlier state that continue as a repeated sequence ($\geq$3\,s), not isolated matches.\\[3pt]
\textbf{Scrambled:} phase-randomized copy, same frequency content, timing destroyed.\\[3pt]
\textbf{Plotted:} the gap DET(real) $-$ DET(scrambled). Mean of 16 videos, 95\% CI; dots = videos.\\[3pt]
DET is a recurrence statistic: 0.93 does not mean the generator is ``93\% deterministic''.
\end{columns}
\vspace{2pt}
{\tiny Per-second DINOv2 [CLS] vectors, 10 principal components per segment; revisit threshold = closest 10\% of state pairs; 5 scrambled copies per segment. Collapse time = when a video stays nearer the shared end state than its opening for 10\,s.\par}
\take{\footnotesize The gap, not the raw score, carries the information. Over whole videos, Self Forcing's raw 0.93 is inflated by its slow collapse and its gap is only 0.04. In matched 60-s windows all three models keep 0.11--0.20 of structure that scrambling removes, and Self Forcing's gap halves after it collapses (0.17 $\to$ 0.09).}
\end{frame}

\begin{frame}{Checking a rival claim: periodic ``snap-back''}
\centering
\figh{0.36}{line_lol}\\[2pt]
{\scriptsize LoL (arXiv 2601.16914) reports LongLive snapping back to its first frames at the same moments in every video. For each video we track sudden rises in similarity to its own first second, average them across the 16 videos of a model, and take the strongest moment (red line). Histogram: the same statistic after shifting each video randomly in time, which destroys any shared timing (500 times).}
\take{No synchronized snap-back within 3 minutes for any model ($p \geq 0.8$). LoL studies hour-long videos, so a longer LongLive run is still open.}
\end{frame}

\begin{frame}{Has anyone done this already?}
\small
About 60 targeted searches of work from 2024 to October 2026. Long-horizon degradation is \textbf{measured widely}. What we did not find is any work that treats streaming video generators as \textbf{dynamical systems}.
\vspace{2pt}
\scriptsize
\renewcommand{\arraystretch}{1.08}
\begin{tabularx}{\linewidth}{>{\raggedright\arraybackslash}p{0.30\linewidth} X >{\raggedright\arraybackslash}p{0.24\linewidth}}
\toprule
Closest work & What it shows & Relation to ours \\
\midrule
VBench-Long, Relax Forcing, In-Distribution Forcing (2610.03120) & Quality, consistency, colour and motion drift over minutes & measure symptoms; we model dynamics \\
LoL (2601.16914) & Sink-collapse loops at hour scale & partial; their videos run an hour, ours 3 min \\
Deep / Rolling Sink, FreqForcing (2607.27110) & A sink slows drift; drift as spectral-energy shift & partial: we show a \emph{type switch} \\
Effective-rank collapse (2607.27036) & Abrupt representation collapse at drift onset & abrupt, where we see a gradual ramp \\
Meta-ARVDM (2503.10704) & Theory of history forgetting & frames, not the prompt \\
Biroli et al., Nat.\ Commun.\ 2024 & Attractor-like regimes \emph{inside} the denoising steps & other time axis \\
Hintze et al., Patterns 2025 & Iterated image loops collapse to $\sim$12 attractors & concept, not video \\
\bottomrule
\end{tabularx}
\vspace{1pt}
\take{\footnotesize Existing work measures long-horizon degradation and specific failure modes. This work characterizes the dynamics: where trajectories converge, how fast prompt information is lost, how perturbations grow, and which inference settings change the long-run regime.}
\end{frame}

\section{Next}

\begin{frame}{Next: planned experiments}
\begin{columns}[T]
\column{0.52\textwidth}
\begin{block}{Generality and statistics}
\begin{itemize}\footnotesize
  \item A second, non-Wan model family
  \item More stochastic realizations: 16--32 prompts $\times$ 4--8 seeds
  \item Compare prompt-conditioned distributions $P(Z_t\mid c)$ directly (MMD, energy distance, prompt decoding)
  \item A second representation (CLIP or SigLIP) beside DINOv2 and the latents
\end{itemize}
\end{block}
\column{0.46\textwidth}
\begin{block}{Dynamics}
\begin{itemize}\footnotesize
  \item Dense sink sweep $s \in \{0,1,2,3,4,6,8\}$; Rolling Forcing with its sink removed
  \item Longer runs: is the late regime stationary? Does a perturbed collapsed state return?
  \item State perturbations swept in size and direction
  \item Running now: drift correction that steers away from the shared end state
\end{itemize}
\end{block}
\end{columns}
\vspace{4pt}
\take{\footnotesize\textbf{The paper in one sentence.} We characterize long-horizon autoregressive video generators as prompt-conditioned stochastic dynamical systems: different generators occupy distinct long-run conditioning regimes, with measurable rates of prompt forgetting, perturbation growth and recurrence structure, and memory settings can move a model between regimes.}
\end{frame}
